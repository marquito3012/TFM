% --- Footer: table with triple-line separator ---
\newcommand{\uaxfooter}{%
  \begin{tabular}[b]{%
    @{}%
    >{\raggedleft\arraybackslash}m{11cm}%
    @{\hspace{0.2cm}}%
    !{\color{uaxblue}\vrule width 0.4pt}%
    @{\hspace{0.8pt}}%
    !{\color{uaxblue}\vrule width 0.4pt}%
    @{\hspace{0.8pt}}%
    !{\color{uaxblue}\vrule width 0.4pt}%
    @{\hspace{0.2cm}}%
    >{\centering\arraybackslash}m{0.8cm}%
    @{}%
  }%
    {\color{uaxdarkblue}\fontsize{12pt}{14pt}\selectfont\textit{\tituloTFM}} &
    {\color{black}\fontsize{12pt}{14pt}\selectfont\thepage}
  \end{tabular}%
}

% --- Reusable header/footer (DRY) ---
\newcommand{\uaxheader}{%
  \fancyhf{}%
  \fancyhead[L]{\textit{\color{uaxdarkblue}\fontsize{12pt}{14pt}\selectfont \autorTFM}}%
  \fancyhead[R]{\raisebox{-0.2cm}{\includegraphics[height=0.93cm]{\logoTFM}}}%
  \fancyfoot[R]{\uaxfooter}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
}
