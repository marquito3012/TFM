% --- Paragraph formatting ---
% Original: Century 12pt, justified, spacing ~24pt before/after, line ~1.15
\setlength{\parindent}{0pt}
\setlength{\parskip}{24pt}
\renewcommand{\baselinestretch}{1.15}

% --- Chapter (Título Capítulo) ---
% Original: 20pt, uaxblue, left-aligned, y starts at 3.50cm from top
\titleformat{\chapter}
  {\fontsize{20pt}{24pt}\selectfont\color{uaxblue}\raggedright}
  {}
  {0pt}
  {}
\titlespacing*{\chapter}{0pt}{-20pt}{36pt}
\makeatletter
\@afterindenttrue
\makeatother

% --- Section (Título Apartado) ---
% Original: 16pt, uaxblue, left-aligned, numbered "1. Title"
\titleformat{\section}
  {\fontsize{16pt}{19.2pt}\selectfont\color{uaxblue}\raggedright}
  {\thesection.}
  {0.4em}
  {}
\titlespacing*{\section}{0pt}{20pt}{12pt}

% --- Subsection (Título Subapartado) ---
% Original: 16pt, uaxblue, left-aligned, numbered "1.1. Title"
\titleformat{\subsection}
  {\fontsize{16pt}{19.2pt}\selectfont\color{uaxblue}\raggedright}
  {\thesubsection.}
  {0.4em}
  {}
\titlespacing*{\subsection}{0pt}{20pt}{12pt}

% --- Section numbering: plain numbers, no chapter prefix ---
\renewcommand{\thesection}{\arabic{section}}
\renewcommand{\thesubsection}{\arabic{section}.\arabic{subsection}}

% --- List styles ---
% Original bullet: 14pt bold serif dash in uaxblue, indent ~1.27cm
\setlist[itemize]{
  label={\fontsize{14pt}{16pt}\selectfont\bfseries\color{uaxblue}--},
  leftmargin=1.27cm,
  topsep=12pt,
  itemsep=6pt,
  parsep=6pt
}
% Original numbered: 12pt bold in uaxblue "1."
\setlist[enumerate]{
  label={\color{uaxblue}\bfseries\arabic*.},
  leftmargin=1.27cm,
  topsep=12pt,
  itemsep=6pt,
  parsep=6pt
}

% --- Bibliography APA (hanging indent 1.25cm) ---
\newenvironment{bibliografiaAPA}{%
  \begin{list}{}{%
    \setlength{\leftmargin}{1.25cm}%
    \setlength{\itemindent}{-1.25cm}%
    \setlength{\parsep}{24pt}%
    \setlength{\itemsep}{0pt}%
    \setlength{\topsep}{0pt}%
  }%
}{%
  \end{list}%
}

% --- Cita Literal APA (block quote, 11pt, indented) ---
\newenvironment{citaLiteralAPA}{%
  \begin{quote}
  \fontsize{11pt}{13.2pt}\selectfont
}{%
  \end{quote}
}

% --- Unnumbered chapter that appears in TOC ---
\newcommand{\chapterUnnumbered}[1]{%
  \chapter*{#1}%
  \addcontentsline{toc}{chapter}{#1}%
  \markboth{#1}{#1}%
}
